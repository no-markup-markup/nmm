# capitalized tokens are probably best handled by the lexer---see separate

# starting point of parsing

nmm-doc := (pars | blks(0)) # TODO: preamble, chs, secs, apps, refs


# what-you-see-is-what-you-get text---up to the intended rare usage of escape
# sequences---without vertical whitespace

txt_no_vws := NON_VWS_CHR+


pars    :=
  par    (NL+         par)* # paragraphs

blks(𝓁) :=
  blk(𝓁) (NL+ 𝓁×TAB blk(𝓁)* # blocks at level 𝓁, with 𝓁 a natural number


# paragraph
# the pilcrow symbol ‘¶’ has unicode code point U+00B6

par      := ‘¶’ SP* (pfx_par | name_par) (NL hdr) NL NL+ blk(0)+

pfx_par  := ‘PAR’ # TODO: more prefixes: ‘DEF’, ‘LMA’, ‘FCT’, ‘RMK’ et cetera

name_par := pfx_par ‘:’ name_sfx

name_sfx := (!(‘[’ / ‘]’ / ‘:’ / ‘,’ / ‘;’) NON_WS_CHR)+


# block

blk(𝓁) := blk_txt(𝓁) | blk_blt(𝓁)


# text block at level 𝓁

blk_txt(𝓁)         := (blk_txt_content(𝓁) NL) (𝓁×TAB blk_txt_content(𝓁) NL)*

blk_txt_content(𝓁) := txt_no_vws # TODO: also txt_emph, c_ref (cross-ref)


# bullet block at level 𝓁

blk_blt(𝓁) := ‘-’ TAB blks(𝓁+1)
